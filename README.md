SELECT  'COKE' AS MATERIAL,
    PERIOD,
    MAX(CASE WHEN FUR_NO = 'C'
             THEN COKE_CONS END) AS C_FCE,
    MAX(CASE WHEN FUR_NO = 'E'
             THEN COKE_CONS END) AS E_FCE,
    MAX(CASE WHEN FUR_NO = 'F'
             THEN COKE_CONS END) AS F_FCE,
             'NUT COKE' AS MATERIAL,
    PERIOD,
    MAX(CASE WHEN FUR_NO = 'C'
             THEN NC_CONS END) AS C_FCE,
    MAX(CASE WHEN FUR_NO = 'E'
             THEN NC_CONS END) AS E_FCE,
    MAX(CASE WHEN FUR_NO = 'F'
             THEN NC_CONS END) AS F_FCE
             
FROM
(
    SELECT FUR_NO, COKE_CONS,NC_CONS,'OnDt' AS PERIOD  
    FROM DEMO.T_BF_PRODUCTION_TEST
    WHERE TIMESTAMP >= TRUNC(SYSDATE) - 2
      AND FUR_NO IN ('C','E','F')

    UNION ALL
    SELECT FUR_NO, COKE_CONS,NC_CONS, 'ToDt' AS PERIOD   
    FROM DEMO.T_BF_PRODUCTION_TODATE_TEST
    WHERE TIMESTAMP >= TRUNC(SYSDATE) - 2
      AND FUR_NO IN ('C','E','F')
)
GROUP BY PERIOD
ORDER BY CASE PERIOD
             WHEN 'OnDt' THEN 1
             WHEN 'ToDt' THEN 2
         END;
