
WITH SOURCE_DATA AS
(
    SELECT
    TIMESTAMP,
        FUR_NO,
        COKE_CONS,
        NC_CONS,
        NOA_ORE_CONS,
        JODA_ORE_CONS,
        SINTER_CONS,
        MN_ORE_CONS,
        QUARTZ_CONS,
        LIMESTONE_CONS,
        DOLOMITE_CONS,
        SCRAP_CONS,
        DUNITE_CONS,
        BREX_CONS,
        DRI_CONS,
        LD_SLAG_CONS,
        PELLET_CONS,
        ASRF_CONS,
        PYROXINITE_CONS,
        COAL_CONS, 
        TAR_CONS,
        'OnDt' AS PERIOD
    FROM DEMO.T_BF_PRODUCTION_TEST
    WHERE TIMESTAMP >= TRUNC(SYSDATE) - 2
      AND FUR_NO IN ('C', 'E', 'F')

    UNION ALL

    SELECT
    TIMESTAMP,
        FUR_NO,
        COKE_CONS,
        NC_CONS,
        NOA_ORE_CONS,
        JODA_ORE_CONS,
        SINTER_CONS,
        MN_ORE_CONS,
        QUARTZ_CONS,
        LIMESTONE_CONS,
        DOLOMITE_CONS,
        SCRAP_CONS,
        DUNITE_CONS,
        BREX_CONS,
        DRI_CONS,
        LD_SLAG_CONS,
        PELLET_CONS,
        ASRF_CONS,
        PYROXINITE_CONS,
        COAL_CONS,
        TAR_CONS,
        'ToDt' AS PERIOD
    FROM DEMO.T_BF_PRODUCTION_TODATE_TEST
    WHERE TIMESTAMP >= TRUNC(SYSDATE) - 2
      AND FUR_NO IN ('C', 'E', 'F')
),
MATERIAL_DATA AS
(
    SELECT 'COKE' AS MATERIAL, PERIOD, FUR_NO,
           COKE_CONS AS CONS
    FROM SOURCE_DATA

    UNION ALL

    SELECT 'NUT COKE' AS MATERIAL, PERIOD, FUR_NO,
           NC_CONS AS CONS
    FROM SOURCE_DATA

    UNION ALL

    SELECT 'Noa Ore' AS MATERIAL, PERIOD, FUR_NO,
           NOA_ORE_CONS AS CONS
    FROM SOURCE_DATA
    UNION ALL
     SELECT 'Joda Ore' AS MATERIAL, PERIOD, FUR_NO,
           JODA_ORE_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Sinter' AS MATERIAL, PERIOD, FUR_NO,
           SINTER_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Mn Ore' AS MATERIAL, PERIOD, FUR_NO,
           MN_ORE_CONS AS CONS
    FROM SOURCE_DATA
      UNION ALL
     SELECT 'Quartz' AS MATERIAL, PERIOD, FUR_NO,
           QUARTZ_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Limestone' AS MATERIAL, PERIOD, FUR_NO,
           LIMESTONE_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Dolomite' AS MATERIAL, PERIOD, FUR_NO,
           DOLOMITE_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Scrap' AS MATERIAL, PERIOD, FUR_NO,
           DOLOMITE_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Dunite' AS MATERIAL, PERIOD, FUR_NO,
           DUNITE_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'Brex' AS MATERIAL, PERIOD, FUR_NO,
           BREX_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'DRI' AS MATERIAL, PERIOD, FUR_NO,
           DRI_CONS AS CONS
    FROM SOURCE_DATA
     UNION ALL
     SELECT 'LD Slag' AS MATERIAL, PERIOD, FUR_NO,
           LD_SLAG_CONS AS CONS
    FROM SOURCE_DATA
    UNION ALL
    SELECT 'Pellet' AS MATERIAL,PERIOD,FUR_NO,
    PELLET_CONS AS CONS From SOURCE_DATA
    UNION ALL
    SELECT 'ASRF' AS MATERIAL,PERIOD,FUR_NO,
    ASRF_CONS AS CONS From SOURCE_DATA
    UNION ALL
    SELECT 'Pyroxinite'AS MATERIAL,PERIOD,FUR_NO,
    PYROXINITE_CONS AS CONS From SOURCE_DATA
    UNION ALL
    SELECT 'Coal Injection' AS MATERIAL,PERIOD,FUR_NO,
    COAL_CONS AS CONS From SOURCE_DATA
    UNION ALL
    SELECT 'Tar Injection' AS MATERIAL,PERIOD,FUR_NO,
    TAR_CONS AS CONS From SOURCE_DATA
)
SELECT TIMESTAMP,
    MATERIAL,
    PERIOD,
    MAX(CASE WHEN FUR_NO = 'C' THEN CONS END) AS CBF,
    MAX(CASE WHEN FUR_NO = 'E' THEN CONS END) AS EBF,
    MAX(CASE WHEN FUR_NO = 'F' THEN CONS END) AS FBF
FROM MATERIAL_DATA
GROUP BY TIMESTAMP,MATERIAL, PERIOD
ORDER BY
    CASE MATERIAL
        WHEN 'COKE' THEN 1
        WHEN 'NUT COKE' THEN 2
        WHEN 'Noa Ore' THEN 3
         WHEN 'Joda Ore' THEN 4
        WHEN 'Sinter'THEN 5
        WHEN 'Mn Ore'THEN 6
        WHEN 'Quartz'Then 7
        WHEN 'Limestone'Then 8
        WHEN 'Dolomite'Then 9
        WHEN 'Scrap'Then 10
        WHEN 'Dunite'Then 11
        WHEN 'Brex'Then 12
        WHEN 'DRI'Then 13
        WHEN 'LD Slag' Then 14
        WHEN 'Pellet' Then 15
        WHEN 'ASRF' Then 16
        WHEN 'Pyroxnite'Then 17
        WHEN 'Coal Injection' Then 18
        WHEN 'Tar Injection'Then 19
        ELSE 20
    END,
    CASE PERIOD
        WHEN 'OnDt' THEN 1
        WHEN 'ToDt' THEN 2
        ELSE 3
    END;
