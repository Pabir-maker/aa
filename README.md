
WITH SOURCE_DATA AS
(
    SELECT
        FUR_NO,
        COKE_CONS,
        NC_CONS,
        NOA_ORE_CONS,
        'OnDt' AS PERIOD
    FROM DEMO.T_BF_PRODUCTION_TEST
    WHERE TIMESTAMP >= TRUNC(SYSDATE) - 2
      AND FUR_NO IN ('C', 'E', 'F')

    UNION ALL

    SELECT
        FUR_NO,
        COKE_CONS,
        NC_CONS,
        NOA_ORE_CONS,
        'ToDt' AS PERIOD
    FROM DEMO.T_BF_PRODUCTION_TODATE_TEST
    WHERE TIMESTAMP >= TRUNC(SYSDATE) - 2
      AND FUR_NO IN ('C', 'E', 'F')
),
MATERIAL_DATA AS
(
    SELECT 'COKE' AS MATERIAL, PERIOD, FUR_NO,
           COKE_CONS AS CONS
    FROM SOURCE_DATA

    UNION ALL

    SELECT 'NUT COKE' AS MATERIAL, PERIOD, FUR_NO,
           NC_CONS AS CONS
    FROM SOURCE_DATA

    UNION ALL

    SELECT 'Noa Ore' AS MATERIAL, PERIOD, FUR_NO,
           NOA_ORE_CONS AS CONS
    FROM SOURCE_DATA
)
SELECT
    MATERIAL,
    PERIOD,
    MAX(CASE WHEN FUR_NO = 'C' THEN CONS END) AS C_FCE,
    MAX(CASE WHEN FUR_NO = 'E' THEN CONS END) AS E_FCE,
    MAX(CASE WHEN FUR_NO = 'F' THEN CONS END) AS F_FCE
FROM MATERIAL_DATA
GROUP BY MATERIAL, PERIOD
ORDER BY
    CASE MATERIAL
        WHEN 'COKE' THEN 1
        WHEN 'NUT COKE' THEN 2
        WHEN 'Noa Ore' THEN 3
        ELSE 4
    END,
    CASE PERIOD
        WHEN 'OnDt' THEN 1
        WHEN 'ToDt' THEN 2
        ELSE 3
    END;
